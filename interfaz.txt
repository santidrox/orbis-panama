# ---------- Utilidades ----------
def now_dir():
    return time.strftime("%Y%m%d_%H%M%S")

def ensure_dir(p):
    os.makedirs(p, exist_ok=True)
    return p

# ---------- Interfaz ----------
@app.get("/")
def ui_root():
    return Response(HTML, mimetype="text/html")

# ---------- Métricas ----------
@app.get("/metrics.json")
def metrics_json():
    return jsonify(LATEST)

@app.post("/ingest")
def ingest():
    try:
        if request.is_json:
            data = request.get_json(force=True)
        else:
            raw = request.data.decode("utf-8", "ignore")
            data = json.loads(raw) if raw else {}
    except Exception as e:
        return jsonify(ok=False, err=f"bad json: {e}"), 400

    for k in ("temp","humidity","pressure","mag_raw","mag_state"):
        if k in data:
            LATEST[k] = data[k]
    LATEST["ts"] = int(time.time())
    return jsonify(ok=True)

# ---------- Cámara: configurar IP ----------
@app.post("/set_cam")
def set_cam():
    global CAM_URL
    try:
        host = (request.get_json(force=True) or {}).get("host","").strip()
    except:
        host = ""
    if not host:
        return jsonify(ok=False, err="host vacío"), 400
    CAM_URL = host if host.startswith("http") else ("http://" + host)
    return jsonify(ok=True, cam=CAM_URL)

# ---------- Proxy: snapshot ----------
@app.get("/cam.jpg")
def cam_jpg():
    if not CAM_URL:
        return ("cam no configurada", 400)
    url = CAM_URL.rstrip("/") + "/jpg"
    try:
        req = Request(url, headers={"User-Agent":"Mozilla/5.0"})
        with urlopen(req, timeout=TIMEOUT) as r:
            data = r.read()
        return Response(data, mimetype="image/jpeg",
                        headers={"Cache-Control":"no-store, no-cache, must-revalidate, max-age=0"})
    except (URLError, HTTPError):
        return ("cam unreachable", 502)

# ---------- Proxy: stream ----------
@app.get("/cam.stream")
def cam_stream():
    if not CAM_URL:
        return ("cam no configurada", 400)
    url = CAM_URL.rstrip("/") + "/stream"
    try:
        req = Request(url, headers={"User-Agent":"Mozilla/5.0"})
        r = urlopen(req, timeout=TIMEOUT)
    except (URLError, HTTPError):
        return ("cam unreachable", 502)

    def gen():
        try:
            while True:
                chunk = r.read(8192)
                if not chunk:
                    break
                yield chunk
        except Exception:
            pass
        finally:
            try: r.close()
            except: pass

    return Response(stream_with_context(gen()),
        mimetype="multipart/x-mixed-replace; boundary=frame",
        headers={"Cache-Control":"no-store, no-cache, must-revalidate, max-age=0"})

# ========== Grabación: hilo que guarda JPGs de /jpg ==========
def _record_worker():
    """Descarga /jpg repetidamente y guarda frame_XXXXXX.jpg en REC['dir']."""
    while not REC["stop_flag"]:
        if not CAM_URL:
            REC["error"] = "cam no configurada"
            break
        try:
            # snapshot
            url = CAM_URL.rstrip("/") + "/jpg"
            req = Request(url, headers={"User-Agent":"Mozilla/5.0"})
            with urlopen(req, timeout=TIMEOUT) as r:
                data = r.read()

            # guarda archivo
            REC["frames"] += 1
            fname = f"frame_{REC['frames']:06d}.jpg"
            with open(os.path.join(REC["dir"], fname), "wb") as f:
                f.write(data)

            # controla tasa ~ 10-15 fps (ajusta a tu gusto: 50-100ms)
            time.sleep(0.16)  # 80 ms ≈ ~12.5 fps

        except Exception as e:
            # si hay un fallo temporal, intenta luego de una pausa corta
            REC["error"] = str(e)
            time.sleep(0.3)

    REC["active"] = False

@app.post("/record/start")
def record_start():
    if REC["active"]:
        return jsonify(ok=False, err="ya está grabando", dir=REC["dir"]), 400
    if not CAM_URL:
        return jsonify(ok=False, err="cam no configurada"), 400

    base = ensure_dir("recordings")
    rec_dir = ensure_dir(os.path.join(base, now_dir()))
    REC.update({
        "active": True,
        "dir": rec_dir,
        "frames": 0,
        "start_ts": time.time(),
        "stop_flag": False,
        "error": None,
        "thread": None,
    })
    th = threading.Thread(target=_record_worker, daemon=True)
    REC["thread"] = th
    th.start()
    return jsonify(ok=True, dir=rec_dir)

@app.post("/record/stop")
def record_stop():
    if not REC["active"]:
        return jsonify(ok=False, err="no estaba grabando"), 400
    REC["stop_flag"] = True
    th = REC.get("thread")
    if th is not None:
        th.join(timeout=3.0)
    dur = max(0.001, time.time() - REC["start_ts"])
    fps = REC["frames"] / dur
    return jsonify(ok=True, dir=REC["dir"], frames=REC["frames"], seconds=round(dur,2), fps=round(fps,2), err=REC["error"])

@app.get("/record/status")
def record_status():
    if not REC["active"]:
        return jsonify(active=False, dir=REC["dir"], frames=REC["frames"], fps=0)
    dur = max(0.001, time.time() - REC["start_ts"])
    fps = REC["frames"] / dur
    return jsonify(active=True, dir=REC["dir"], frames=REC["frames"], fps=round(fps,2), err=REC["error"])

# ---------- Salud ----------
@app.get("/health")
def health():

# ---------- Catch-all GET -> UI ----------
@app.route("/<path:_x>", methods=["GET"])
def catch_all(_x):
    if _x in ("metrics.json","ingest","set_cam","cam.jpg","cam.stream","record/start","record/stop","record/status","health"):
        return ("Not Found", 404)
    return Response(HTML, mimetype="text/html")
